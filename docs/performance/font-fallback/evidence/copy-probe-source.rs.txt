//! Standalone allocator/copy investigation, not an acceptance benchmark.
//! No zpdf code is called, and original Criterion evidence remains unchanged.

use std::hint::black_box;
use std::io::{BufWriter, Write};
use std::time::{Duration, Instant};

const INPUT: &[u8] = include_bytes!("../crates/zpdf-font/tests/fixtures/coverage-primary.ttf");
const ITERATIONS: usize = 50_000;
const BATCHES: usize = 32;

#[link(name = "System")]
unsafe extern "C" {
    fn pthread_self() -> usize;
    fn pthread_get_qos_class_np(thread: usize, class: *mut u32, priority: *mut i32) -> i32;
    fn pthread_set_qos_class_self_np(class: u32, priority: i32) -> i32;
}

fn main() {
    let mode = std::env::args().nth(1).unwrap_or_else(|| "normal".into());
    if mode == "user_initiated" {
        assert_eq!(unsafe { pthread_set_qos_class_self_np(0x19, 0) }, 0);
    } else {
        assert_eq!(mode, "normal");
    }
    let mut qos = 0;
    let mut priority = 0;
    assert_eq!(unsafe { pthread_get_qos_class_np(pthread_self(), &mut qos, &mut priority) }, 0);
    assert_eq!(INPUT.len(), 21_088);
    let source = INPUT.to_vec();
    let mut target = vec![0u8; INPUT.len()];
    let mut output = BufWriter::new(std::io::stdout().lock());
    writeln!(output, "{{\"kind\":\"environment\",\"mode\":\"{mode}\",\"qos\":{qos},\"iterations\":{ITERATIONS},\"batches\":{BATCHES},\"bytes\":{}}}", INPUT.len()).unwrap();
    for stage in ["allocate", "copy_preallocated", "clone_heap"] {
        let warm = Instant::now();
        while warm.elapsed() < Duration::from_millis(200) {
            for _ in 0..1000 {
                match stage {
                    "allocate" => { black_box(Vec::<u8>::with_capacity(black_box(INPUT.len()))); }
                    "copy_preallocated" => { target.copy_from_slice(black_box(&source)); black_box(&target); }
                    _ => { black_box(black_box(source.as_slice()).to_vec()); }
                }
            }
        }
        for batch in 0..BATCHES {
            let mut addresses = [0usize; 64];
            let start = Instant::now();
            for iteration in 0..ITERATIONS {
                match stage {
                    "allocate" => {
                        let value = Vec::<u8>::with_capacity(black_box(INPUT.len()));
                        if iteration < addresses.len() { addresses[iteration] = value.as_ptr() as usize; }
                        black_box(value);
                    }
                    "copy_preallocated" => {
                        target.copy_from_slice(black_box(&source));
                        if iteration < addresses.len() { addresses[iteration] = target.as_ptr() as usize; }
                        black_box(&target);
                    }
                    _ => {
                        let value = black_box(source.as_slice()).to_vec();
                        if iteration < addresses.len() { addresses[iteration] = value.as_ptr() as usize; }
                        black_box(value);
                    }
                }
            }
            let ns = start.elapsed().as_nanos() as f64 / ITERATIONS as f64;
            addresses.sort_unstable();
            let unique = 1 + addresses.windows(2).filter(|pair| pair[0] != pair[1]).count();
            writeln!(output, "{{\"kind\":\"batch\",\"stage\":\"{stage}\",\"batch\":{batch},\"ns_per_op\":{ns},\"source_mod_16k\":{},\"destination_mod_16k\":{},\"sampled_destinations\":{unique}}}", source.as_ptr() as usize % 16_384, addresses[0] % 16_384).unwrap();
        }
        if stage == "copy_preallocated" { assert_eq!(target, source); }
    }
}
