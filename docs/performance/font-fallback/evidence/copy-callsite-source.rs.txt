//! Independent allocation-callsite probe; not a zpdf acceptance benchmark.
use std::hint::black_box;
use std::time::{Duration, Instant};

const FONT: &[u8] = include_bytes!("../../crates/zpdf-font/tests/fixtures/coverage-primary.ttf");
const ITERATIONS: u64 = 131_072;

#[inline(never)]
#[no_mangle]
pub fn copy_shared_site(bytes: &[u8]) -> Vec<u8> {
    bytes.to_vec()
}

#[inline(never)]
#[no_mangle]
pub fn copy_distinct_site(bytes: &[u8]) -> Vec<u8> {
    let copy = bytes.to_vec();
    // Keep this untimed helper distinct under identical-function folding.
    black_box(copy.len());
    copy
}

fn main() {
    let mode = std::env::args().nth(1).expect("explicit allocation mode");
    assert!(mode == "shared" || mode == "distinct");
    assert_eq!(FONT.len(), 21_088);
    eprintln!("probe identity: {}", env!("ZPDF_COPY_PROBE_ID"));
    let source = if mode == "shared" {
        copy_shared_site(black_box(FONT))
    } else {
        copy_distinct_site(black_box(FONT))
    };
    assert_eq!(source.as_slice(), FONT);
    let copied = copy_shared_site(black_box(&source));
    assert_eq!(copied.as_slice(), FONT);
    drop(copied);
    let warm = Instant::now();
    while warm.elapsed() < Duration::from_millis(100) {
        for _ in 0..4096 {
            drop(black_box(copy_shared_site(black_box(&source))));
        }
    }
    let mut samples = Vec::with_capacity(10);
    for _ in 0..10 {
        let start = Instant::now();
        for _ in 0..ITERATIONS {
            drop(black_box(copy_shared_site(black_box(&source))));
        }
        samples.push(start.elapsed().as_nanos());
    }
    println!("{{\"mode\":\"{mode}\",\"iterations\":{ITERATIONS},\"samples_ns\":{samples:?}}}");
}
